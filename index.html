function addProject(){ var company = document.getElementById('p-company').value.trim(); var date = document.getElementById('p-date').value; var cpw = parseFloat(document.getElementById('p-cpw').value); var wc = parseFloat(document.getElementById('p-wc').value); var sc = document.getElementById('p-sc').value.trim(); var dia = document.getElementById('p-dia').value.trim(); var depth = document.getElementById('p-depth').value.trim(); // 🔴 DOĞRU VALIDASYON if(company === '' || date === '' || isNaN(cpw) || isNaN(wc)){ notify('Tüm zorunlu alanları doldurun!','var(--red)'); return; } if(editingProjectId){ var idx = projects.findIndex(p => p.id === editingProjectId); if(idx !== -1){ projects[idx].company = company; projects[idx].date = date; projects[idx].cementPerWell = cpw; projects[idx].wellCount = wc; projects[idx].totalCement = cpw * wc; projects[idx].scRatio = sc; projects[idx].diameter = dia; projects[idx].depth = depth; } editingProjectId = null; document.getElementById('p-form-title').textContent='Yeni Proje'; document.getElementById('p-save-btn').textContent='+ Proje Ekle'; document.getElementById('p-save-btn').style.background=''; save(); refreshAll(); clearProjectForm(); notify('Proje güncellendi!','var(--greenL)'); } else { projects.push({ id: Date.now(), company, date, cementPerWell: cpw, wellCount: wc, totalCement: cpw * wc, scRatio: sc, diameter: dia, depth: depth, createdAt: new Date().toISOString() }); save(); refreshAll(); clearProjectForm(); notify('Proje eklendi!','var(--greenL)'); } }
